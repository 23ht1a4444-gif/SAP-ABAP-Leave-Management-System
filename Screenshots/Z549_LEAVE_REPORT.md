*&---------------------------------------------------------------------*
*& Report Z549_LEAVE_REPORT
*&---------------------------------------------------------------------*
*& Leave Request Report
*&---------------------------------------------------------------------*
REPORT Z549_LEAVE_REPORT.

TABLES: ZLEAVE_REQ0549.

SELECT-OPTIONS:
  S_EMPID FOR ZLEAVE_REQ0549-EMPLOYEE_ID,
  S_FROM  FOR ZLEAVE_REQ0549-FROM_DATE,
  S_TO    FOR ZLEAVE_REQ0549-TO_DATE,
  S_STAT  FOR ZLEAVE_REQ0549-STATUS.

DATA:
  LT_REQ TYPE TABLE OF ZLEAVE_REQ0549,
  LS_REQ TYPE ZLEAVE_REQ0549.

DATA:
  FCAT TYPE SLIS_T_FIELDCAT_ALV,
  WCAT TYPE SLIS_FIELDCAT_ALV.

START-OF-SELECTION.

  SELECT *
    FROM ZLEAVE_REQ0549
    INTO TABLE LT_REQ
    WHERE EMPLOYEE_ID IN S_EMPID
      AND FROM_DATE   IN S_FROM
      AND TO_DATE     IN S_TO
      AND STATUS      IN S_STAT.

  IF LT_REQ IS INITIAL.
    MESSAGE 'No Leave Requests Found' TYPE 'I'.
    RETURN.
  ENDIF.

  WCAT-FIELDNAME = 'REQST_ID'.
  WCAT-SELTEXT_M = 'Request ID'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'EMPLOYEE_ID'.
  WCAT-SELTEXT_M = 'Employee ID'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'LEAVE_TYPE'.
  WCAT-SELTEXT_M = 'Leave Type'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'FROM_DATE'.
  WCAT-SELTEXT_M = 'From Date'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'TO_DATE'.
  WCAT-SELTEXT_M = 'To Date'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'NO_OF_DAYS'.
  WCAT-SELTEXT_M = 'No Of Days'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'REASON'.
  WCAT-SELTEXT_M = 'Reason'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'STATUS'.
  WCAT-SELTEXT_M = 'Status'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'APPROVED_BY'.
  WCAT-SELTEXT_M = 'Approved By'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'CREATED_ON'.
  WCAT-SELTEXT_M = 'Created On'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      IT_FIELDCAT = FCAT
    TABLES
      T_OUTTAB    = LT_REQ.
