*&---------------------------------------------------------------------*
*& Report ZLEAVE_BALANCE
*&---------------------------------------------------------------------*
*& Leave Balance Report
*&---------------------------------------------------------------------*
REPORT ZLEAVE_BALANCE.

TABLES: ZLEV_BALENCE0549.

SELECT-OPTIONS:
  S_EMPID FOR ZLEV_BALENCE0549-EMPLOYEE_ID,
  S_LEAVE FOR ZLEV_BALENCE0549-LEAVE_TYPE.

DATA:
  LT_BAL TYPE TABLE OF ZLEV_BALENCE0549,
  LS_BAL TYPE ZLEV_BALENCE0549.

DATA:
  FCAT TYPE SLIS_T_FIELDCAT_ALV,
  WCAT TYPE SLIS_FIELDCAT_ALV.

DATA:
  LAYOUT TYPE SLIS_LAYOUT_ALV.

START-OF-SELECTION.

  SELECT *
    FROM ZLEV_BALENCE0549
    INTO TABLE LT_BAL
    WHERE EMPLOYEE_ID IN S_EMPID
      AND LEAVE_TYPE  IN S_LEAVE.

  IF LT_BAL IS INITIAL.
    MESSAGE 'No Leave Balance Found' TYPE 'I'.
    RETURN.
  ENDIF.

  WCAT-FIELDNAME = 'EMPLOYEE_ID'.
  WCAT-SELTEXT_M = 'Employee ID'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'LEAVE_TYPE'.
  WCAT-SELTEXT_M = 'Leave Type'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'TOTAL_DAYS'.
  WCAT-SELTEXT_M = 'Total Days'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'USED_DAYS'.
  WCAT-SELTEXT_M = 'Used Days'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  WCAT-FIELDNAME = 'BALANCE_DAYS'.
  WCAT-SELTEXT_M = 'Balance Days'.
  APPEND WCAT TO FCAT.
  CLEAR WCAT.

  LAYOUT-ZEBRA = 'X'.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      IT_FIELDCAT = FCAT
       IS_LAYOUT   = LAYOUT
    TABLES
      T_OUTTAB    = LT_BAL.
